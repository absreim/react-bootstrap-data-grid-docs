# Layout

The _grid_ component allows one to customize the width of individual columns as well as both the width and height of the
entire grid.

## Column Width

Column widths are set as a fixed number of pixels as part of the definition of each column. If no width is specified, it
defaults to 100 pixels.

## Row Height

There are no row height settings. Row height is always fixed to contain one line of text.

More specifically, row height is determined by font size and cell padding, which are in turn set by
[Bootstrap variables](https://getbootstrap.com/docs/5.3/customize/sass/#variable-defaults).
By default, font size is `1rem` and padding on both top and bottom is `0.5rem` each, resulting in total row height of
`2rem`.

## Grid Dimensions

There are three settings for both width and height when it comes to setting dimensions for the entire grid:

<dl>
    <dt>Auto</dt>
    <dd>Sets width or height according the intrinsic size of the contents of the grid. This is the default setting.</dd>
    <dt>Parent</dt>
    <dd>Sets the width or height to be the 100% of the width or height of the parent container.</dd>
    <dt>Fixed</dt>
    <dd>Sets the width of height to a specified number of pixels.</dd>
</dl>

For the _Parent_ or _Fixed_ setting, the grid becomes scrollable in the relevant dimension if there is content that
overflows.

## Main Component Spacing

Enabling certain features, like [pagination](/docs/grid/pagination), [filtering](/docs/overview/filtering), or
[export](/docs/overview/export), causes other components to be displayed alongside the grid.

The spacing between these components can be configured via the [styleOptions](/docs/ref/grid-specific-props) prop to
the [Grid](/docs/ref/grid) component. See the [relevant example](#main-component-spacing-example) below to see this
setting in action.

```tsx
"use client";

import Grid, {
  ColDef,
  GridProps,
  RowDef,
} from "@absreim/react-bootstrap-data-grid/grid";
import { FC, useMemo } from "react";
import Stack from "react-bootstrap/Stack";
import { OptionSet } from "@/assets/examples/types";
import useOptionSelection from "@/assets/examples/useOptionSelection";

// --- Selectable width and height options UI ---

type ColWidthOptionNames = "none" | "100px" | "200px";
type GridDimOptionNames = "none" | "auto" | "parent" | "fixed";

const colWidthOptions: OptionSet<ColWidthOptionNames, ColDef["width"]> = {
  none: {
    label: "None (same as 100px)",
    value: undefined,
  },
  "100px": {
    label: "100px",
    value: 100,
  },
  "200px": {
    label: "200px",
    value: 200,
  },
};

const gridWidthOptions: OptionSet<GridDimOptionNames, GridProps["width"]> = {
  none: {
    label: 'None (same as "auto")',
    value: undefined,
  },
  auto: {
    label: "Auto",
    value: "auto",
  },
  parent: {
    label: "Parent",
    value: "parent",
  },
  fixed: {
    label: "Fixed (250px)",
    value: 250,
  },
};

const gridHeightOptions: OptionSet<GridDimOptionNames, GridProps["height"]> = {
  none: {
    label: 'None (same as "auto")',
    value: undefined,
  },
  auto: {
    label: "Auto",
    value: "auto",
  },
  parent: {
    label: "Parent",
    value: "parent",
  },
  fixed: {
    label: "Fixed (200px)",
    value: 200,
  },
};

// --- Sample data ---

interface CompanyInfo {
  name: string;
  marketCap: number;
}

const rows: RowDef<CompanyInfo>[] = [
  {
    id: "nvda",
    data: {
      name: "Nvidia",
      marketCap: 4.9,
    },
  },
  {
    id: "goog",
    data: {
      name: "Alphabet",
      marketCap: 4.7,
    },
  },
  {
    id: "aapl",
    data: {
      name: "Apple",
      marketCap: 4.2,
    },
  },
  {
    id: "msft",
    data: {
      name: "Microsoft",
      marketCap: 3.0,
    },
  },
  {
    id: "amzn",
    data: {
      name: "Amazon",
      marketCap: 2.9,
    },
  },
  {
    id: "meta",
    data: {
      name: "Meta Platforms",
      marketCap: 1.5,
    },
  },
  {
    id: "tsla",
    data: {
      name: "Tesla",
      marketCap: 1.5,
    },
  },
];

const getCols: (widthSetting: ColDef["width"]) => ColDef[] = (widthSetting) => [
  {
    name: "name",
    type: "string",
    label: "Company Name",
    width: widthSetting,
  },
  {
    name: "marketCap",
    type: "number",
    label: "Market Cap (USD, trillions)",
    width: widthSetting,
  },
];

// --- Demo component ---

const LayoutSettingsDemo: FC = () => {
  const colWidthOptionHook = useOptionSelection(
    colWidthOptions,
    "none",
    "Column Width Settings",
    "colWidth",
  );
  const gridWidthOptionsHook = useOptionSelection(
    gridWidthOptions,
    "none",
    "Grid Width Settings",
    "gridWidth",
  );
  const gridHeightOptionsHook = useOptionSelection(
    gridHeightOptions,
    "none",
    "Grid Height Settings",
    "gridHeight",
  );

  const cols = useMemo(
    () => getCols(colWidthOptionHook.value),
    [colWidthOptionHook.value],
  );

  return (
    <Stack gap={2}>
      <Stack gap={2}>
        {colWidthOptionHook.selectionUi}
        {gridWidthOptionsHook.selectionUi}
        {gridHeightOptionsHook.selectionUi}
      </Stack>
      <div
        style={{
          width: 300,
          height: 300,
        }}
        className="overflow-auto border"
      >
        <Grid
          rows={rows}
          cols={cols}
          width={gridWidthOptionsHook.value}
          height={gridHeightOptionsHook.value}
          borders="full"
          borderVariant="primary"
        />
      </div>
    </Stack>
  );
};

export default LayoutSettingsDemo;
```

```tsx
"use client";

import Grid, { ColDef, RowDef } from "@absreim/react-bootstrap-data-grid/grid";
import planetData from "./planets.json";
import { Planet } from "./types";
import { FC, ReactNode, useState } from "react";
import Form from "react-bootstrap/Form";

// --- begin data for grid ---
// Data source: Endless Space 2 Fandom

export const cols: ColDef[] = [
    { type: "string", name: "planet", label: "Planet", width: 140 },
    { type: "string", name: "climate", label: "Climate", width: 100 },
    { type: "string", name: "fertility", label: "Fertility", width: 100 },
    {
        type: "number",
        name: "minPopulation",
        label: "Min Population",
        width: 140,
    },
    {
        type: "number",
        name: "maxPopulation",
        label: "Max Population",
        width: 140,
    },
    { type: "number", name: "approval", label: "Approval", width: 100 },
    { type: "number", name: "food", label: "Food", width: 80 },
    { type: "number", name: "industry", label: "Industry", width: 100 },
    { type: "number", name: "dust", label: "Dust", width: 80 },
    { type: "number", name: "science", label: "Science", width: 100 },
    { type: "number", name: "influence", label: "Influence", width: 100 },
    { type: "string", name: "requiredTech", label: "Required Tech", width: 200 },
    {
        type: "string",
        name: "riftbornRequiredTech",
        label: "Riftborn Required Tech",
        width: 230,
    },
    { type: "string", name: "description", label: "Description", width: 700 },
];

const rows: RowDef<Planet>[] = planetData.map((planet) => ({
    id: planet.planet,
    data: planet,
}));

// --- end data for grid ---

// --- start form options setup ---

type GapOptions = Record<string, { label: string; value: number | undefined }>;

const gapOptions: GapOptions = [0, 1, 2, 3, 4, 5].reduce(
    (prev, curr) => {
        prev[String(curr)] = {
            label: String(curr),
            value: curr,
        };
        return prev;
    },
    {
        unset: {
            label: "Unset (same as 2)",
            value: undefined,
        },
    } as GapOptions,
);
const gapOptNameToLabel = Object.keys(gapOptions).reduce(
    (prev, curr) => {
        prev[curr] = gapOptions[curr].label;
        return prev;
    },
    {} as Record<string, string>,
);

function getRadioGroup(
    legend: string,
    optionLabels: Record<string, string>,
    currentVal: string,
    setVal: (val: string) => void,
    settingTypeId: string,
): ReactNode {
    return (
        <fieldset key={settingTypeId}>
            <legend>{legend}</legend>
            {Object.keys(optionLabels).map((optionName) => {
                return (
                    <Form.Check
                        key={optionName}
                        type="radio"
                        id={`${settingTypeId}-${optionName}`}
                        label={optionLabels[optionName]}
                        checked={currentVal === optionName}
                        value={optionName}
                        onChange={({ target }) => setVal(target.value)}
                    />
                );
            })}
        </fieldset>
    );
}

// --- end form options setup ---

const MainComponentLayoutDemo: FC = () => {
    const [mainSpacing, setMainSpacing] = useState<string>("unset");

    return (
        <>
            <Form>
                {getRadioGroup(
                    "Main Component Spacing",
                    gapOptNameToLabel,
                    mainSpacing,
                    setMainSpacing,
                    "mainSpacingSetting",
                )}
            </Form>
            <Grid
                rows={rows}
                cols={cols}
                width="parent"
                styleOptions={{ mainComponentSpacing: gapOptions[mainSpacing].value }}
                pagination={{ type: "uncontrolled" }}
                allowExport
                filterModel={{ type: "uncontrolled" }}
            />
        </>
    );
}

export default MainComponentLayoutDemo;
```
